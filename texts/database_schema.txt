1. ENABLE EXTENSIONS
================================================================================

-- Enable UUID generation (usually enabled by default in Supabase)
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

================================================================================
2. JOBS TABLE
================================================================================

-- Main table for storing job applications
CREATE TABLE jobs (
    -- Primary key: UUID auto-generated
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    
    -- Foreign key: links to authenticated user
    -- ON DELETE CASCADE ensures jobs are deleted when user is deleted
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE NOT NULL,
    
    -- Required fields
    company VARCHAR(255) NOT NULL,
    role VARCHAR(255) NOT NULL,
    
    -- Optional salary range (stored as integers, in thousands or annual)
    salary_min INTEGER CHECK (salary_min >= 0),
    salary_max INTEGER CHECK (salary_max >= 0),
    
    -- Optional job posting URL
    url TEXT,
    
    -- Status with enum constraint
    -- DEFAULT 'wishlist' for new entries
    status VARCHAR(50) DEFAULT 'wishlist' CHECK (
        status IN ('wishlist', 'applied', 'interviewing', 'offer', 'rejected')
    ),
    
    -- Date when application was submitted (NULL if not yet applied)
    date_applied DATE,
    
    -- Free-form notes field
    notes TEXT,
    
    -- Timestamps
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    
    -- Constraint: salary_max should be >= salary_min when both are provided
    CONSTRAINT valid_salary_range CHECK (
        salary_min IS NULL OR salary_max IS NULL OR salary_max >= salary_min
    )
);

-- Add comments for documentation
COMMENT ON TABLE jobs IS 'Stores job applications tracked by users';
COMMENT ON COLUMN jobs.status IS 'Application status: wishlist, applied, interviewing, offer, rejected';
COMMENT ON COLUMN jobs.salary_min IS 'Minimum salary (annual, in user preferred currency)';
COMMENT ON COLUMN jobs.salary_max IS 'Maximum salary (annual, in user preferred currency)';

================================================================================
3. INDEXES FOR PERFORMANCE
================================================================================

-- Index on user_id for filtering by user (most common query pattern)
CREATE INDEX idx_jobs_user_id ON jobs(user_id);

-- Index on status for filtering by status
CREATE INDEX idx_jobs_status ON jobs(status);

-- Index on date_applied for time-based queries and sorting
CREATE INDEX idx_jobs_date_applied ON jobs(date_applied);

-- Composite index for common query: user's jobs by status
CREATE INDEX idx_jobs_user_status ON jobs(user_id, status);

-- Index for company search (case-insensitive)
CREATE INDEX idx_jobs_company_lower ON jobs(LOWER(company));

================================================================================
4. ROW LEVEL SECURITY (RLS)
================================================================================

-- Enable RLS on the jobs table
ALTER TABLE jobs ENABLE ROW LEVEL SECURITY;

-- Policy: Users can only SELECT their own jobs
CREATE POLICY "Users can view own jobs" 
    ON jobs
    FOR SELECT 
    USING (auth.uid() = user_id);

-- Policy: Users can only INSERT jobs for themselves
CREATE POLICY "Users can insert own jobs" 
    ON jobs
    FOR INSERT 
    WITH CHECK (auth.uid() = user_id);

-- Policy: Users can only UPDATE their own jobs
CREATE POLICY "Users can update own jobs" 
    ON jobs
    FOR UPDATE 
    USING (auth.uid() = user_id)
    WITH CHECK (auth.uid() = user_id);

-- Policy: Users can only DELETE their own jobs
CREATE POLICY "Users can delete own jobs" 
    ON jobs
    FOR DELETE 
    USING (auth.uid() = user_id);

================================================================================
5. TRIGGERS
================================================================================

-- Function to automatically update the updated_at timestamp
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger to call the function before each UPDATE
CREATE TRIGGER set_updated_at
    BEFORE UPDATE ON jobs
    FOR EACH ROW
    EXECUTE FUNCTION update_updated_at_column();

8. FIELD DESCRIPTIONS
================================================================================

| Field        | Type         | Required | Description                        |
|--------------|--------------|----------|------------------------------------|
| id           | UUID         | Auto     | Unique identifier                  |
| user_id      | UUID         | Yes      | Owner of the job entry             |
| company      | VARCHAR(255) | Yes      | Company name                       |
| role         | VARCHAR(255) | Yes      | Job title/position                 |
| salary_min   | INTEGER      | No       | Minimum salary (annual)            |
| salary_max   | INTEGER      | No       | Maximum salary (annual)            |
| url          | TEXT         | No       | Link to job posting                |
| status       | VARCHAR(50)  | Yes*     | Current application status         |
| date_applied | DATE         | No       | When application was submitted     |
| notes        | TEXT         | No       | Additional notes                   |
| created_at   | TIMESTAMPTZ  | Auto     | When record was created            |
| updated_at   | TIMESTAMPTZ  | Auto     | When record was last modified      |

* status defaults to 'wishlist' if not provided

| applied      | Application submitted, waiting to hear back   | Blue    |

| offer        | Received an offer                              | Green   |
